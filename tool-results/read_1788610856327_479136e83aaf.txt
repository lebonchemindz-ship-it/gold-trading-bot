   132→   132→function wilsonLower(wins: number, n: number, z = 1.96): number {
   133→   133→  if (n <= 0) return 0;
   134→   134→  const p = wins / n;
   135→   135→  const denom = 1 + (z * z) / n;
   136→   136→  const center = p + (z * z) / (2 * n);
   137→   137→  const spread = z * Math.sqrt((p * (1 - p)) / n + (z * z) / (4 * n * n));
   138→   138→  return Math.max(0, (center - spread) / denom);
   139→   139→}

   156→   156→function statsOf(trades: RawTrade[]): BtStats {
   157→   157→  const wins = trades.filter((t) => t.result === "win");
   158→   158→  const losses = trades.filter((t) => t.result === "loss");
   159→   159→  const timeouts = trades.filter((t) => t.result === "timeout");
   160→   160→  const grossR = trades.reduce((a, t) => a + t.r, 0);
   161→   161→  const posR = trades.filter((t) => t.r > 0).reduce((a, t) => a + t.r, 0);
   162→   162→  const negR = Math.abs(trades.filter((t) => t.r < 0).reduce((a, t) => a + t.r, 0));
   163→   163→
   164→   164→  // أقصى تراجع تراكمي
   165→   165→  let cum = 0;
   166→   166→  let peak = 0;
   167→   167→  let maxDd = 0;
   168→   168→  for (const t of trades) {
   169→   169→    cum += t.r;
   170→   170→    peak = Math.max(peak, cum);
   171→   171→    maxDd = Math.max(maxDd, peak - cum);
   172→   172→  }
   173→   173→
   174→   174→  // السلاسل
   175→   175→  let best = 0;
   176→   176→  let worst = 0;
   177→   177→  let curW = 0;
   178→   178→  let curL = 0;
   179→   179→  for (const t of trades) {
   180→   180→    if (t.result === "win") {
   181→   181→      curW++;
   182→   182→      curL = 0;
   183→   183→    } else {
   184→   184→      curL++;
   185→   185→      curW = 0;
   186→   186→    }
   187→   187→    best = Math.max(best, curW);
   188→   188→    worst = Math.max(worst, curL);
   189→   189→  }
   190→   190→
   191→   191→  const n = trades.length;
   192→   192→  return {
   193→   193→    trades: n,
   194→   194→    wins: wins.length,
   195→   195→    losses: losses.length,
   196→   196→    timeouts: timeouts.length,
   197→   197→    winRate: n ? wins.length / n : 0,
   198→   198→    avgR: n ? grossR / n : 0,
   199→   199→    grossR: Math.round(grossR * 100) / 100,
   200→   200→    profitFactor: negR > 0 ? Math.round((posR / negR) * 100) / 100 : posR > 0 ? 99 : 0,
   201→   201→    expectancyR: n ? Math.round((grossR / n) * 1000) / 1000 : 0,
   202→   202→    maxDrawdownR: Math.round(maxDd * 100) / 100,
   203→   203→    bestStreak: best,
   204→   204→    worstStreak: worst,
   205→   205→  };
   206→   206→}

   208→   208→/** محاكاة النتيجة — الوقف له الأولوية داخل الشمعة (متحفظ) */
   209→   209→function simulate(
   210→   210→  candles: Candle[],
   211→   211→  i: number,
   212→   212→  dir: 1 | -1,
   213→   213→  sl: number,
   214→   214→  hold: number
   215→   215→): { result: "win" | "loss" | "timeout"; r: number; bars: number; endIdx: number } | null {
   216→   216→  const next = candles[i + 1];
   217→   217→  if (!next) return null;
   218→   218→  const entry = next.o;
   219→   219→  const risk = Math.abs(entry - sl);
   220→   220→  if (risk < 0.3) return null; // وقف غير منطقي — تجاهل
   221→   221→  const tp1 = entry + dir * risk;
   222→   222→
   223→   223→  const last = Math.min(i + hold, candles.length - 1);
   224→   224→  for (let j = i + 1; j <= last; j++) {
   225→   225→    const c = candles[j];
   226→   226→    if (dir === 1) {
   227→   227→      if (c.l <= sl) return { result: "loss", r: -1, bars: j - i, endIdx: j };
   228→   228→      if (c.h >= tp1) return { result: "win", r: 1, bars: j - i, endIdx: j };
   229→   229→    } else {
   230→   230→      if (c.h >= sl) return { result: "loss", r: -1, bars: j - i, endIdx: j };
   231→   231→      if (c.l <= tp1) return { result: "win", r: 1, bars: j - i, endIdx: j };
   232→   232→    }
   233→   233→  }
   234→   234→  const exit = candles[last].c;
   235→   235→  const r = ((exit - entry) * dir) / risk;
   236→   236→  return { result: "timeout", r: Math.round(r * 100) / 100, bars: last - i, endIdx: last };
   237→   237→}

   318→   318→function evaluateAt(ctx: Ctx, i: number): Vote[] {
   319→   319→  const votes: Vote[] = [];
   320→   320→  const { candles, closes } = ctx;
   321→   321→  const c = candles[i];
   322→   322→  const p = candles[i - 1];
   323→   323→  const a = ctx.atrArr[i];
   324→   324→  if (!a || a < 0.3 || isNaN(a)) return votes;
   325→   325→
   326→   326→  const body = Math.abs(c.c - c.o);
   327→   327→  const e9 = ctx.ema9[i];
   328→   328→  const e21 = ctx.ema21[i];
   329→   329→  const e50 = ctx.ema50[i];
   330→   330→  const e200 = ctx.ema200[i];
   331→   331→  const rsiNow = ctx.rsiArr[i];
   332→   332→  const rsiPrev = ctx.rsiArr[i - 1];
   333→   333→
   334→   334→  const ok = (...vals: number[]) => vals.every((v) => v != null && !isNaN(v));
   335→   335→
   336→   336→  // 1) ema_trend: ترتيب كامل + ADX ≥ 20
   337→   337→  if (ok(e9, e21, e50, ctx.adxArr[i]) && ctx.adxArr[i] >= 20) {
   338→   338→    if (e9 > e21 && e21 > e50 && c.c > e9) votes.push({ key: "ema_trend", dir: 1, sl: c.c - 1.5 * a });
   339→   339→    if (e9 < e21 && e21 < e50 && c.c < e9) votes.push({ key: "ema_trend", dir: -1, sl: c.c + 1.5 * a });
   340→   340→  }
   341→   341→
   342→   342→  // 2) macd_cross: تقاطع خلال آخر شمعتين + اتجاه EMA50
   343→   343→  const ml = ctx.macdLine[i];
   344→   344→  const ms = ctx.macdSignal[i];
   345→   345→  const mlP = ctx.macdLine[i - 1];
   346→   346→  const msP = ctx.macdSignal[i - 1];
   347→   347→  if (ok(ml, ms, mlP, msP, e50)) {
   348→   348→    if (ml > ms && mlP <= msP && c.c > e50) votes.push({ key: "macd_cross", dir: 1, sl: c.c - 1.5 * a });
   349→   349→    if (ml < ms && mlP >= msP && c.c < e50) votes.push({ key: "macd_cross", dir: -1, sl: c.c + 1.5 * a });
   350→   350→  }
   351→   351→
   352→   352→  // 3) bb_break: انضغاط سابق + إغلاق خارج الشريط بجسم قوي
   353→   353→  const bwNow = ctx.bw[i];
   354→   354→  const bwAvgNow = ctx.bwAvg[i];
   355→   355→  const squeezePrev = ok(ctx.bw[i - 1], ctx.bwAvg[i - 1]) && ctx.bw[i - 1] < 0.8 * ctx.bwAvg[i - 1];
   356→   356→  if (squeezePrev && ok(ctx.bbUpper[i], ctx.bbLower[i]) && body > 0.5 * a) {
   357→   357→    if (c.c > ctx.bbUpper[i]) votes.push({ key: "bb_break", dir: 1, sl: c.c - 1.5 * a });
   358→   358→    if (c.c < ctx.bbLower[i]) votes.push({ key: "bb_break", dir: -1, sl: c.c + 1.5 * a });
   359→   359→  }
   360→   360→  void bwNow;
   361→   361→  void bwAvgNow;
   362→   362→
   363→   363→  // 4) rsi_reversal: خروج من التطرف + مع EMA200
   364→   364→  if (ok(rsiNow, rsiPrev, e200)) {
   365→   365→    if (rsiPrev < 32 && rsiNow > rsiPrev && c.c > e200)
   366→   366→      votes.push({ key: "rsi_reversal", dir: 1, sl: c.c - 1.5 * a });
   367→   367→    if (rsiPrev > 68 && rsiNow < rsiPrev && c.c < e200)
   368→   368→      votes.push({ key: "rsi_reversal", dir: -1, sl: c.c + 1.5 * a });
   369→   369→  }
   370→   370→
   371→   371→  // 5) bb_reversal: اختراق الشريط ثم عودة داخلية + RSI
   372→   372→  const up = ctx.bbUpper[i];
   373→   373→  const lo = ctx.bbLower[i];
   374→   374→  const upP = ctx.bbUpper[i - 1];
   375→   375→  const loP = ctx.bbLower[i - 1];
   376→   376→  if (ok(up, lo, upP, loP, rsiNow)) {
   377→   377→    if (p.l < loP && c.c > lo && rsiNow < 42)
   378→   378→      votes.push({ key: "bb_reversal", dir: 1, sl: c.c - 1.5 * a });
   379→   379→    if (p.h > upP && c.c < up && rsiNow > 58)
   380→   380→      votes.push({ key: "bb_reversal", dir: -1, sl: c.c + 1.5 * a });
   381→   381→  }
   382→   382→
   383→   383→  // 6) sr_bounce: دعم/مقاومة ديناميكي (آخر 12 شمعة سابقة)
   384→   384→  let support = Infinity;
   385→   385→  let resistance = -Infinity;

   387→   387→    if (j < 0) continue;
   388→   388→    support = Math.min(support, candles[j].l);
   389→   389→    resistance = Math.max(resistance, candles[j].h);
   390→   390→  }
   391→   391→  if (ok(support, resistance) && isFinite(support) && isFinite(resistance)) {
   392→   392→    if (c.l <= support + 0.25 * a && c.c > c.o && c.c > support)
   393→   393→      votes.push({ key: "sr_bounce", dir: 1, sl: c.c - 1.2 * a });
   394→   394→    if (c.h >= resistance - 0.25 * a && c.c < c.o && c.c < resistance)
   395→   395→      votes.push({ key: "sr_bounce", dir: -1, sl: c.c + 1.2 * a });
   396→   396→  }
   397→   397→
   398→   398→  // 7) smc_sweep: كنس قاع/قمة اليوم السابق ثم استعادة
   399→   399→  const dayIdx = ctx.dayOf[i];
   400→   400→  if (dayIdx >= 1) {
   401→   401→    const prevDay = ctx.days[dayIdx - 1];
   402→   402→    if (prevDay && prevDay.lastIdx < i) {
   403→   403→      if (c.l < prevDay.low && c.c > prevDay.low && c.c > c.o) {
   404→   404→        const slDist = clamp(c.c - c.l + 0.25 * a, 0.8 * a, 2.5 * a);
   405→   405→        votes.push({ key: "smc_sweep", dir: 1, sl: c.c - slDist });
   406→   406→      }
   407→   407→      if (c.h > prevDay.high && c.c < prevDay.high && c.c < c.o) {
   408→   408→        const slDist = clamp(c.h - c.c + 0.25 * a, 0.8 * a, 2.5 * a);
   409→   409→        votes.push({ key: "smc_sweep", dir: -1, sl: c.c + slDist });
   410→   410→      }
   411→   411→    }
   412→   412→  }
   413→   413→
   414→   414→  // 8) pattern_play: ابتلاع/مطرقة/شهاب مع فلتر EMA200
   415→   415→  const bodyP = c.h - c.l > 0 ? body / (c.h - c.l) : 0;
   416→   416→  const upperWick = c.h - Math.max(c.c, c.o);
   417→   417→  const lowerWick = Math.min(c.c, c.o) - c.l;
   418→   418→  const prevBody = Math.abs(p.c - p.o);
   419→   419→  if (ok(e200) && c.h - c.l > 0) {
   420→   420→    const bullEngulf = p.c < p.o && c.c > c.o && c.c >= p.o && c.o <= p.c && body > prevBody;
   421→   421→    const bearEngulf = p.c > p.o && c.c < c.o && c.o >= p.c && c.c <= p.o && body > prevBody;
   422→   422→    const hammer = lowerWick > body * 2 && upperWick < body * 0.8 && bodyP < 0.45;
   423→   423→    const star = upperWick > body * 2 && lowerWick < body * 0.8 && bodyP < 0.45;
   424→   424→    if ((bullEngulf || (hammer && c.c > c.o)) && c.c > e200)
   425→   425→      votes.push({ key: "pattern_play", dir: 1, sl: c.c - 1.2 * a });
   426→   426→    if ((bearEngulf || (star && c.c < c.o)) && c.c < e200)
   427→   427→      votes.push({ key: "pattern_play", dir: -1, sl: c.c + 1.2 * a });
   428→   428→  }
   429→   429→
   430→   430→  // 9) asian_breakout: كسر نطاق آسيوي بعد 07:00 UTC
   431→   431→  const dayInfo = ctx.days[dayIdx];
   432→   432→  const hourUtc = new Date(c.t).getUTCHours();
   433→   433→  if (dayInfo && dayInfo.asianCount >= 4 && hourUtc >= 7 && isFinite(dayInfo.asianHigh) && isFinite(dayInfo.asianLow)) {
   434→   434→    const rangeWidth = dayInfo.asianHigh - dayInfo.asianLow;
   435→   435→    const onlyAfterLondon = hourUtc >= 7 && hourUtc <= 16; // نافذة لندن فقط
   436→   436→    if (onlyAfterLondon && body > 0.6 * a) {
   437→   437→      if (c.c > dayInfo.asianHigh) {
   438→   438→        const slDist = Math.max(1.0 * a, rangeWidth * 0.5);
   439→   439→        votes.push({ key: "asian_breakout", dir: 1, sl: c.c - slDist });
   440→   440→      }
   441→   441→      if (c.c < dayInfo.asianLow) {
   442→   442→        const slDist = Math.max(1.0 * a, rangeWidth * 0.5);
   443→   443→        votes.push({ key: "asian_breakout", dir: -1, sl: c.c + slDist });
   444→   444→      }
   445→   445→    }
   446→   446→  }
   447→   447→
   448→   448→  return votes;
   449→   449→}

   451→   451→// ---------- التعلّم ----------
   452→   452→function learnWeights(
   453→   453→  stratTrades: Map<string, RawTrade[]>,
   454→   454→  base: Record<string, number>
   455→   455→): Record<string, number> {
   456→   456→  const mult: Record<string, number> = {};
   457→   457→  for (const meta of STRATEGY_META) {
   458→   458→    const trades = stratTrades.get(meta.key) ?? [];
   459→   459→    const n = trades.length;
   460→   460→    if (n >= 12) {
   461→   461→      const wins = trades.filter((t) => t.result === "win").length;
   462→   462→      const wl = wilsonLower(wins, n);
   463→   463→      mult[meta.key] = clamp(wl / 0.5, 0.55, 1.45);
   464→   464→    } else {
   465→   465→      mult[meta.key] = 1; // عينة صغيرة — لا تعلّم
   466→   466→    }
   467→   467→  }
   468→   468→
   469→   469→  // مضاعف العمود = متوسط مرجّح بعدد صفقات استراتيجياته
   470→   470→  const pillarMult: Record<string, number> = {};
   471→   471→  for (const pKey of PILLAR_KEYS) {
   472→   472→    let num = 0;
   473→   473→    let den = 0;
   474→   474→    for (const meta of STRATEGY_META) {
   475→   475→      if (meta.pillar !== pKey) continue;
   476→   476→      const n = (stratTrades.get(meta.key) ?? []).length;
   477→   477→      num += mult[meta.key] * n;
   478→   478→      den += n;
   479→   479→    }
   480→   480→    pillarMult[pKey] = den > 0 ? num / den : 1;
   481→   481→  }
   482→   482→
   483→   483→  const learned: Record<string, number> = {};
   484→   484→  for (const pKey of PILLAR_KEYS) {
   485→   485→    learned[pKey] = base[pKey] * pillarMult[pKey];
   486→   486→  }
   487→   487→  // إعادة التطبيع إلى 100 مع حد أدنى 6
   488→   488→  let sum = PILLAR_KEYS.reduce((a, k) => a + learned[k], 0);
   489→   489→  for (const k of PILLAR_KEYS) learned[k] = Math.max(6, learned[k]);
   490→   490→  sum = PILLAR_KEYS.reduce((a, k) => a + learned[k], 0);
   491→   491→  for (const k of PILLAR_KEYS) learned[k] = Math.round((learned[k] / sum) * 1000) / 10;
   492→   492→  return learned;
   493→   493→}

   496→   496→export async function runBacktest(tf: "15m" | "1h"): Promise<BacktestResult> {
   497→   497→  const interval = tf === "1h" ? "60m" : "15m";
   498→   498→  const range = tf === "1h" ? "3mo" : "60d";
   499→   499→  const hold = tf === "1h" ? 24 : 48; // يوم تداول / 12 ساعة
   500→   500→
   501→   501→  const candles = await fetchCandles("GC=F", interval, range);
   502→   502→  if (candles.length < WARMUP + 400) {
   503→   503→    throw new Error(`بيانات غير كافية للتدريب (${candles.length} شمعة) — جرّب الإطار الآخر`);
   504→   504→  }
   505→   505→
   506→   506→  const closes = candles.map((c) => c.c);
   507→   507→  const emaSet = computeEMASet(closes);
   508→   508→  const rsiArr = rsi(closes, 14);
   509→   509→  const macdData = macd(closes);
   510→   510→  const bbData = bollinger(closes);
   511→   511→  const atrData = atr(candles, 14);
   512→   512→
   513→   513→  // سلسلة ADX سببية (Wilder) — نفس رياضيات indicators.ts لكن لكل شمعة
   514→   514→  const adxArr = adxSeries(candles, 14);
   515→   515→
   516→   516→  // عرض بولينجر + متوسطه المتحرك السببي
   517→   517→  const bw: number[] = new Array(candles.length).fill(NaN);
   518→   518→  for (let i = 0; i < candles.length; i++) {
   519→   519→    const u = bbData.upper[i];
   520→   520→    const m = bbData.mid[i];
   521→   521→    if (!isNaN(u) && !isNaN(m) && m > 0) bw[i] = ((bbData.upper[i] - bbData.lower[i]) / m) * 100;
   522→   522→  }
   523→   523→  const bwAvg = rollingAvg(bw, 50);
   524→   524→
   525→   525→  const { days, dayOf } = buildDays(candles);
   526→   526→
   527→   527→  const ctx: Ctx = {
   528→   528→    candles,
   529→   529→    closes,
   530→   530→    ema9: emaSet.ema9,
   531→   531→    ema21: emaSet.ema21,
   532→   532→    ema50: emaSet.ema50,
   533→   533→    ema200: emaSet.ema200,
   534→   534→    rsiArr,
   535→   535→    macdLine: macdData.line,
   536→   536→    macdSignal: macdData.signal,
   537→   537→    macdHist: macdData.histogram,
   538→   538→    bbUpper: bbData.upper,
   539→   539→    bbLower: bbData.lower,
   540→   540→    bw,
   541→   541→    bwAvg,
   542→   542→    atrArr: atrData.values,
   543→   543→    adxArr,
   544→   544→    days,
   545→   545→    dayOf,
   546→   546→  };
   547→   547→
   548→   548→  const len = candles.length;
   549→   549→  const split = WARMUP + Math.floor((len - WARMUP - hold) * 0.7); // 70% تدريب / 30% اختبار
   550→   550→
   551→   551→  // ===== 1) تقييم كل الشموع (التصويت) — مرة واحدة =====
   552→   552→  const votesAll: Vote[][] = new Array(len);
   553→   553→  for (let i = WARMUP; i < len - 1; i++) {
   554→   554→    votesAll[i] = evaluateAt(ctx, i);
   555→   555→  }
   556→   556→
   557→   557→  // ===== 2) صفقات كل استراتيجية (مع تهدئة — صفقة واحدة نشطة) =====
   558→   558→  const stratTrades = new Map<string, RawTrade[]>();
   559→   559→  const takenPerStrategy = new Map<string, number[]>(); // فهرس الشموع المأخوذة
   560→   560→  for (const meta of STRATEGY_META) {
   561→   561→    const trades: RawTrade[] = [];
   562→   562→    const taken: number[] = [];
   563→   563→    let busyUntil = -1;
   564→   564→    let lastAsianDay = -1; // كسر آسيوي واحد لكل يوم
   565→   565→    for (let i = WARMUP; i < len - 1; i++) {
   566→   566→      const votes = votesAll[i] ?? [];
   567→   567→      const vote = votes.find((v) => v.key === meta.key);
   568→   568→      if (!vote) continue;
   569→   569→      if (i <= busyUntil) continue;
   570→   570→      if (meta.key === "asian_breakout") {
   571→   571→        const d = dayOf[i];
   572→   572→        if (d === lastAsianDay) continue;
   573→   573→        lastAsianDay = d;
   574→   574→      }
   575→   575→      const sim = simulate(candles, i, vote.dir, vote.sl, hold);
   576→   576→      if (!sim) continue;
   577→   577→      trades.push({
   578→   578→        key: meta.key,
   579→   579→        i,
   580→   580→        t: candles[i].t,
   581→   581→        entryT: candles[i + 1].t,
   582→   582→        dir: vote.dir,
   583→   583→        entry: candles[i + 1].o,
   584→   584→        sl: vote.sl,
   585→   585→        tp1: candles[i + 1].o + vote.dir * Math.abs(candles[i + 1].o - vote.sl),
   586→   586→        result: sim.result,
   587→   587→        r: sim.r,
   588→   588→        bars: sim.bars,
   589→   589→        endIdx: sim.endIdx,
   590→   590→      });
   591→   591→      taken.push(i);
   592→   592→      busyUntil = sim.endIdx;
   593→   593→    }
   594→   594→    stratTrades.set(meta.key, trades);
   595→   595→    takenPerStrategy.set(meta.key, taken);
   596→   596→  }

   598→   598→  // ===== 3) التوليفة المركّبة (إشارة البوت الكلية) =====
   599→   599→  function runComposite(
   600→   600→    from: number,
   601→   601→    to: number,
   602→   602→    weights: Record<string, number> | null // null = أوزان متساوية
   603→   603→  ): RawTrade[] {
   604→   604→    const trades: RawTrade[] = [];
   605→   605→    let busyUntil = -1;
   606→   606→    let lastAsianDay = -1;
   607→   607→    for (let i = from; i < to; i++) {
   608→   608→      const votes = votesAll[i] ?? [];
   609→   609→      if (!votes.length) continue;
   610→   610→      if (i <= busyUntil) continue;
   611→   611→      let num = 0;
   612→   612→      let den = 0;
   613→   613→      for (const v of votes) {
   614→   614→        const meta = STRATEGY_META.find((m) => m.key === v.key)!;
   615→   615→        const w = weights ? weights[meta.pillar] : 1;
   616→   616→        num += v.dir * w;
   617→   617→        den += w;
   618→   618→      }
   619→   619→      if (den === 0) continue;
   620→   620→      const net = num / den;
   621→   621→      if (net < 0.15 && net > -0.15) continue;
   622→   622→      const dir: 1 | -1 = net >= 0.15 ? 1 : -1;
   623→   623→
   624→   624→      const agreeing = votes.filter((v) => v.dir === dir);
   625→   625→      if (agreeing.length < 2) continue; // توافق صورتين على الأقل
   626→   626→
   627→   627→      // كسر آسيوي واحد لكل يوم (لتناسق الاستراتيجيات)
   628→   628→      const hasAsian = agreeing.some((v) => v.key === "asian_breakout");
   629→   629→      if (hasAsian) {
   630→   630→        const d = dayOf[i];
   631→   631→        if (d === lastAsianDay) continue;
   632→   632→        lastAsianDay = d;
   633→   633→      }
   634→   634→
   635→   635→      const c = candles[i];
   636→   636→      const a = ctx.atrArr[i];
   637→   637→      const sl = dir === 1 ? c.c - 1.5 * a : c.c + 1.5 * a;
   638→   638→      const sim = simulate(candles, i, dir, sl, hold);
   639→   639→      if (!sim) continue;
   640→   640→      trades.push({
   641→   641→        key: "composite",
   642→   642→        i,
   643→   643→        t: candles[i].t,
   644→   644→        entryT: candles[i + 1].t,
   645→   645→        dir,
   646→   646→        entry: candles[i + 1].o,
   647→   647→        sl,
   648→   648→        tp1: candles[i + 1].o + dir * Math.abs(candles[i + 1].o - sl),
   649→   649→        result: sim.result,
   650→   650→        r: sim.r,
   651→   651→        bars: sim.bars,
   652→   652→        endIdx: sim.endIdx,
   653→   653→      });
   654→   654→      busyUntil = sim.endIdx;
   655→   655→    }
   656→   656→    return trades;
   657→   657→  }
   658→   658→
   659→   659→  // ===== 4) التعلم على فترة التدريب ثم الاختبار الإسقاطي =====
   660→   660→  const trainStratTrades = new Map<string, RawTrade[]>();
   661→   661→  for (const [k, ts] of stratTrades) trainStratTrades.set(k, ts.filter((t) => t.i < split));
   662→   662→
   663→   663→  const base = { ...BASE_PILLAR_WEIGHTS } as Record<string, number>;
   664→   664→  const learnedTestW = learnWeights(trainStratTrades, base);
   665→   665→
   666→   666→  const tradesBeforeTest = runComposite(split, len - 1, null); // أوزان متساوية
   667→   667→  const tradesAfterTest = runComposite(split, len - 1, learnedTestW); // أوزان متعلمة
   668→   668→  const tradesTrainSeg = runComposite(WARMUP, split, null); // داخل عينة التدريب
   669→   669→
   670→   670→  // أوزان التطبيق المباشر — تعلم من كامل التاريخ
   671→   671→  const fullStratTrades = stratTrades;
   672→   672→  const learnedLiveW = learnWeights(fullStratTrades, base);
   673→   673→  const tradesLive = runComposite(WARMUP, len - 1, learnedLiveW);
   674→   674→
   675→   675→  // حفظ حالة التعلّم ليمتد إليها محرك الإشارة المباشر
   676→   676→  setLearnedState({
   677→   677→    weights: learnedLiveW,
   678→   678→    stats: {
   679→   679→      winRate: tradesLive.length ? tradesLive.filter((t) => t.result === "win").length / tradesLive.length : 0,
   680→   680→      trades: tradesLive.length,
   681→   681→      profitFactor: statsOf(tradesLive).profitFactor,
   682→   682→      expectancyR: statsOf(tradesLive).expectancyR,
   683→   683→      maxDrawdownR: statsOf(tradesLive).maxDrawdownR,
   684→   684→      tf,
   685→   685→      from: new Date(candles[WARMUP].t).toISOString(),
   686→   686→      to: new Date(candles[len - 1].t).toISOString(),
   687→   687→      updatedAt: new Date().toISOString(),
   688→   688→    },
   689→   689→    tf,
   690→   690→    ts: Date.now(),
   691→   691→  });
   692→   692→
   693→   693→  // ===== 5) نتائج الاستراتيجيات (كامل الفترة) =====
   694→   694→  const strategies: BtStrategy[] = STRATEGY_META.map((meta) => {
   695→   695→    const ts = stratTrades.get(meta.key) ?? [];
   696→   696→    const st = statsOf(ts);
   697→   697→    const wins = st.wins;
   698→   698→    const n = st.trades;
   699→   699→    const wl = wilsonLower(wins, n);
   700→   700→    return {
   701→   701→      key: meta.key,
   702→   702→      nameAr: meta.nameAr,
   703→   703→      pillar: meta.pillar,
   704→   704→      pillarAr: meta.pillarAr,
   705→   705→      trades: n,
   706→   706→      wins,
   707→   707→      losses: st.losses,
   708→   708→      timeouts: st.timeouts,
   709→   709→      winRate: Math.round(st.winRate * 1000) / 10,
   710→   710→      wilson: Math.round(wl * 1000) / 10,
   711→   711→      avgR: Math.round(st.avgR * 100) / 100,
   712→   712→      profitFactor: st.profitFactor,
   713→   713→      multiplier: n >= 12 ? Math.round(clamp(wl / 0.5, 0.55, 1.45) * 100) / 100 : 1,
   714→   714→      grossR: st.grossR,
   715→   715→    };
   716→   716→  });
   717→   717→
   718→   718→  // ===== 6) منحنى R التراكمي =====
   719→   719→  const equity: number[] = [0];
   720→   720→  for (const t of tradesAfterTest) equity.push(Math.round((equity[equity.length - 1] + t.r) * 100) / 100);
   721→   721→  const equityBefore: number[] = [0];
   722→   722→  for (const t of tradesBeforeTest) equityBefore.push(Math.round((equityBefore[equityBefore.length - 1] + t.r) * 100) / 100);
   723→   723→
   724→   724→  // ===== 7) تحليل الشهور (الشهر الحالي + السابق) =====
   725→   725→  const months: BtMonth[] = [];
   726→   726→  {
   727→   727→    interface MonthAgg {
   728→   728→      ym: string;
   729→   729→      open: number;
   730→   730→      close: number;
   731→   731→      high: number;
   732→   732→      low: number;
   733→   733→      atrSum: number;
   734→   734→      atrN: number;
   735→   735→    }
   736→   736→    const byMonth = new Map<string, MonthAgg>();
   737→   737→    const order: string[] = [];
   738→   738→    for (let i = WARMUP; i < len; i++) {
   739→   739→      const d = new Date(candles[i].t);
   740→   740→      const ym = `${d.getUTCFullYear()}-${String(d.getUTCMonth() + 1).padStart(2, "0")}`;
   741→   741→      let agg = byMonth.get(ym);
   742→   742→      if (!agg) {
   743→   743→        agg = { ym, open: candles[i].o, close: candles[i].c, high: -Infinity, low: Infinity, atrSum: 0, atrN: 0 };
   744→   744→        byMonth.set(ym, agg);
   745→   745→        order.push(ym);
   746→   746→      }
   747→   747→      agg.close = candles[i].c;
   748→   748→      agg.high = Math.max(agg.high, candles[i].h);
   749→   749→      agg.low = Math.min(agg.low, candles[i].l);
   750→   750→      const a = ctx.atrArr[i];
   751→   751→      if (!isNaN(a) && a > 0) {
   752→   752→        agg.atrSum += a;
   753→   753→        agg.atrN++;
   754→   754→      }
   755→   755→    }
   756→   756→    for (const ym of order.slice(-3)) {
   757→   757→      const agg = byMonth.get(ym)!;
   758→   758→      const mt = tradesLive.filter((t) => {
   759→   759→        const d = new Date(t.t);
   760→   760→        return `${d.getUTCFullYear()}-${String(d.getUTCMonth() + 1).padStart(2, "0")}` === ym;
   761→   761→      });
   762→   762→      const mWins = mt.filter((t) => t.result === "win").length;
   763→   763→      const buys = mt.filter((t) => t.dir === 1).length;

   847→   847→function rollingAvg(values: number[], win: number): number[] {
   848→   848→  const out: number[] = new Array(values.length).fill(NaN);
   849→   849→  let sum = 0;
   850→   850→  let count = 0;
   851→   851→  const queue: number[] = [];
   852→   852→  for (let i = 0; i < values.length; i++) {
   853→   853→    const v = values[i];
   854→   854→    if (!isNaN(v)) {
   855→   855→      queue.push(v);
   856→   856→      sum += v;
   857→   857→      count++;
   858→   858→      if (queue.length > win) {
   859→   859→        sum -= queue.shift()!;
   860→   860→        count--;
   861→   861→      }
   862→   862→    }
   863→   863→    out[i] = count >= Math.min(win, 20) ? sum / count : NaN;
   864→   864→  }
   865→   865→  return out;
   866→   866→}

   869→   869→function adxSeries(candles: Candle[], period: number): number[] {
   870→   870→  const n = candles.length;
   871→   871→  const out: number[] = new Array(n).fill(NaN);
   872→   872→  if (n < period * 2 + 1) return out;
   873→   873→
   874→   874→  const plusDM: number[] = [0];
   875→   875→  const minusDM: number[] = [0];
   876→   876→  const trs: number[] = [candles[0].h - candles[0].l];
   877→   877→  for (let i = 1; i < n; i++) {
   878→   878→    const upMove = candles[i].h - candles[i - 1].h;
   879→   879→    const downMove = candles[i - 1].l - candles[i].l;
   880→   880→    plusDM.push(upMove > downMove && upMove > 0 ? upMove : 0);
   881→   881→    minusDM.push(downMove > upMove && downMove > 0 ? downMove : 0);
   882→   882→    const prevClose = candles[i - 1].c;
   883→   883→    trs.push(Math.max(candles[i].h - candles[i].l, Math.abs(candles[i].h - prevClose), Math.abs(candles[i].l - prevClose)));
   884→   884→  }
   885→   885→
   886→   886→  const smooth = (arr: number[]): number[] => {
   887→   887→    const s: number[] = new Array(arr.length).fill(NaN);
   888→   888→    let sum = 0;
   889→   889→    for (let i = 0; i < period; i++) sum += arr[i];
   890→   890→    s[period - 1] = sum;
   891→   891→    for (let i = period; i < arr.length; i++) s[i] = s[i - 1] - s[i - 1] / period + arr[i];
   892→   892→    return s;
   893→   893→  };
   894→   894→
   895→   895→  const smTR = smooth(trs);
   896→   896→  const smPDM = smooth(plusDM);
   897→   897→  const smMDM = smooth(minusDM);
   898→   898→
   899→   899→  const dxs: number[] = [];
   900→   900→  const dxIdx: number[] = []; // فهرس الشمعة المقابل
   901→   901→  for (let i = period - 1; i < n; i++) {
   902→   902→    const tr = smTR[i];
   903→   903→    if (tr > 0 && !isNaN(tr)) {
   904→   904→      const pdi = (smPDM[i] / tr) * 100;
   905→   905→      const mdi = (smMDM[i] / tr) * 100;
   906→   906→      const sum = pdi + mdi;
   907→   907→      dxs.push(sum > 0 ? (Math.abs(pdi - mdi) / sum) * 100 : 0);
   908→   908→      dxIdx.push(i);
   909→   909→    }
   910→   910→  }
   911→   911→
   912→   912→  if (dxs.length >= period) {
   913→   913→    let a = 0;
   914→   914→    for (let i = 0; i < period; i++) a += dxs[i];
   915→   915→    a /= period;
   916→   916→    out[dxIdx[period - 1]] = a;
   917→   917→    for (let i = period; i < dxs.length; i++) {
   918→   918→      a = (a * (period - 1) + dxs[i]) / period;
   919→   919→      out[dxIdx[i]] = a;
   920→   920→    }
   921→   921→  }
   922→   922→  return out;
   923→   923→}
