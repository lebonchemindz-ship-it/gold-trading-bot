   171→/** يوم بيانات عالي التأثير؟ (NFP أول جمعة + FOMC + CPI تقديري) */
   172→function highImpactDay(t: number): string | null {
   173→  const d = new Date(t);
   174→  const y = d.getUTCFullYear();
   175→  const m = d.getUTCMonth();
   176→  const day = d.getUTCDate();
   177→  const key = `${y}-${pad2(m + 1)}-${pad2(day)}`;
   178→  if (FOMC_DAYS.has(key)) return "اجتماع الفيدرالي FOMC (قرار الفائدة)";
   179→  // أول جمعة من الشهر = NFP (يوم الإصدار أو اليوم السابق)
   180→  const firstDow = new Date(Date.UTC(y, m, 1)).getUTCDay();
   181→  const firstFriday = ((5 - firstDow + 7) % 7) + 1;
   182→  if (day === firstFriday || day === firstFriday - 1) return "بيانات الوظائف الأمريكية NFP (تقديري)";
   183→  if (day >= 11 && day <= 13) return "بيانات التضخم CPI (تقديري)";
   184→  return null;
   185→}

   187→/** تصنيف نظام السوق عند الشمعة i (سببي — لا مستقبل) */
   188→function regimeAt(ctx: Ctx, i: number): Regime {
   189→  const adx = ctx.adxArr[i];
   190→  const e50 = ctx.ema50[i];
   191→  const e200 = ctx.ema200[i];
   192→  if (isNaN(adx) || isNaN(e50) || isNaN(e200)) return "range";
   193→  if (adx >= 20) return e50 >= e200 ? "trend_up" : "trend_down";
   194→  return "range";
   195→}

   199→  i: number;

   215→function statsOfTrades(trades: TrainerTrade[]): BtStats {
   216→  const wins = trades.filter((t) => t.result === "win").length;
   217→  const losses = trades.filter((t) => t.result === "loss").length;
   218→  const timeouts = trades.length - wins - losses;
   219→  const grossR = trades.reduce((a, t) => a + t.r, 0);
   220→  const posR = trades.filter((t) => t.r > 0).reduce((a, t) => a + t.r, 0);
   221→  const negR = Math.abs(trades.filter((t) => t.r < 0).reduce((a, t) => a + t.r, 0));
   222→
   223→  let cum = 0;
   224→  let peak = 0;
   225→  let maxDd = 0;
   226→  for (const t of trades) {
   227→    cum += t.r;
   228→    peak = Math.max(peak, cum);
   229→    maxDd = Math.max(maxDd, peak - cum);
   230→  }
   231→  let best = 0;
   232→  let worst = 0;
   233→  let curW = 0;
   234→  let curL = 0;
   235→  for (const t of trades) {
   236→    if (t.result === "win") {
   237→      curW++;
   238→      curL = 0;
   239→    } else {
   240→      curL++;
   241→      curW = 0;
   242→    }
   243→    best = Math.max(best, curW);
   244→    worst = Math.max(worst, curL);
   245→  }
   246→  const n = trades.length;
   247→  return {
   248→    trades: n,
   249→    wins,
   250→    losses,
   251→    timeouts,
   252→    winRate: n ? wins / n : 0,
   253→    avgR: n ? grossR / n : 0,
   254→    grossR: r2(grossR),
   255→    profitFactor: negR > 0 ? r2(posR / negR) : posR > 0 ? 99 : 0,
   256→    expectancyR: n ? r3(grossR / n) : 0,
   257→    maxDrawdownR: r2(maxDd),
   258→    bestStreak: best,
   259→    worstStreak: worst,
   260→  };
   261→}
   262→
   263→function equityOf(trades: TrainerTrade[]): number[] {
   264→  const eq: number[] = [0];
   265→  for (const t of trades) eq.push(r2(eq[eq.length - 1] + t.r));
   266→  return eq;
   267→}

   272→export async function runSelfTraining(requestedEpochs: number): Promise<SelfTrainingResult> {
   273→  const epochsWanted = clamp(Math.floor(requestedEpochs), 2, MAX_EPOCHS);
   274→
   275→  // ---------- 1) البيانات: ساعة × 6 أشهر (احتياطي: يومي × سنتين) ----------
   276→  let candles: Candle[] = [];
   277→  let dataSource = "";
   278→  let tf: "1h" | "1d" = "1h";
   279→  let hold = 24; // بالشموع
   280→
   281→  try {
   282→    candles = await fetchCandles("GC=F", "60m", "6mo");
   283→    dataSource = "Yahoo Finance GC=F — 6 أشهر @ شمعة الساعة";
   284→  } catch {
   285→    candles = [];
   286→  }
   287→
   288→  let startIdx = WARMUP_1H;
   289→  if (candles.length < 800) {
   290→    // وضع احتياطي: يومي — نستخدم آخر ~126 يوم تداول (≈6 أشهر)
   291→    const daily = await fetchCandles("GC=F", "1d", "2y");
   292→    if (daily.length < 260) {
   293→      throw new Error(`بيانات غير كافية للتدريب (${daily.length} شمعة يومية)`);
   294→    }
   295→    candles = daily;
   296→    tf = "1d";
   297→    hold = 10;
   298→    startIdx = Math.max(40, candles.length - 130);
   299→    dataSource = "Yahoo Finance GC=F — وضع يومي احتياطي (آخر 6 أشهر)";
   300→  }
   301→
   302→  const len = candles.length;
   303→  if (len - startIdx < 60) throw new Error("نافذة التدريب ضيقة جداً — أعد المحاولة");
   304→
   305→  // ---------- 2) المؤشرات السببية (مرة واحدة) ----------
   306→  const closes = candles.map((c) => c.c);
   307→  const emaSet = computeEMASet(closes);
   308→  const rsiArr = rsi(closes, 14);
   309→  const macdData = macd(closes);
   310→  const bbData = bollinger(closes);
   311→  const atrData = atr(candles, 14);
   312→  const adxArr = adxSeries(candles, 14);
   313→  const atrAvg = rollingAvg(atrData.values, 50);
   314→
   315→  const bw: number[] = new Array(len).fill(NaN);
   316→  for (let i = 0; i < len; i++) {
   317→    const u = bbData.upper[i];
   318→    const m = bbData.mid[i];
   319→    if (!isNaN(u) && !isNaN(m) && m > 0) bw[i] = ((bbData.upper[i] - bbData.lower[i]) / m) * 100;
   320→  }
   321→  const bwAvg = rollingAvg(bw, 50);
   322→  const { days, dayOf } = buildDays(candles);
   323→
   324→  const ctx: Ctx = {
   325→    candles,
   326→    closes,
   327→    ema9: emaSet.ema9,
   328→    ema21: emaSet.ema21,
   329→    ema50: emaSet.ema50,
   330→    ema200: emaSet.ema200,
   331→    rsiArr,
   332→    macdLine: macdData.line,
   333→    macdSignal: macdData.signal,
   334→    macdHist: macdData.histogram,
   335→    bbUpper: bbData.upper,
   336→    bbLower: bbData.lower,
   337→    bw,
   338→    bwAvg,
   339→    atrArr: atrData.values,
   340→    adxArr,
   341→    days,
   342→    dayOf,
   343→  };
   344→
   345→  // ---------- 3) تصويت الاستراتيجيات عند كل شمعة (مستقل عن الأوزان) ----------
   346→  const votesAll: Vote[][] = new Array(len);
   347→  for (let i = startIdx; i < len - 1; i++) {
   348→    votesAll[i] = evaluateAt(ctx, i);
   349→  }
   350→  const regimes: Regime[] = new Array(len);
   351→  for (let i = startIdx; i < len; i++) regimes[i] = regimeAt(ctx, i);
   352→
   353→  // ذاكرة أيام الأحداث (لكل مفتاح يوم)
   354→  const eventByDay = new Map<string, string | null>();
   355→  function eventAt(i: number): string | null {
   356→    const d = new Date(candles[i].t);
   357→    const k = `${d.getUTCFullYear()}-${pad2(d.getUTCMonth() + 1)}-${pad2(d.getUTCDate())}`;
   358→    if (!eventByDay.has(k)) eventByDay.set(k, highImpactDay(candles[i].t));
   359→    return eventByDay.get(k) ?? null;
   360→  }
   361→
   362→  // ---------- 4) التوليفة الموزونة بالأوزان والمضاعفات ----------
   363→  type MultMap = Record<string, Record<Regime, number>>;
   364→  const initMults = (): MultMap => {
   365→    const m: MultMap = {};
   366→    for (const meta of STRATEGY_META) m[meta.key] = { trend_up: 1, trend_down: 1, range: 1 };
   367→    return m;
   368→  };
   369→
   370→  function runComposite(
   371→    from: number,
   372→    to: number,
   373→    pillarW: Record<string, number>,
   374→    mults: MultMap
   375→  ): TrainerTrade[] {
   376→    const trades: TrainerTrade[] = [];
   377→    let busyUntil = -1;
   378→    let lastAsianDay = -1;
   379→    for (let i = from; i < to; i++) {
   380→      if (i <= busyUntil) continue;
   381→      const votes = votesAll[i] ?? [];
   382→      if (!votes.length) continue;
   383→      const reg = regimes[i];
   384→      let num = 0;
   385→      let den = 0;
   386→      for (const v of votes) {
   387→        const meta = META_BY_KEY.get(v.key);
   388→        if (!meta) continue;
   389→        const w = (pillarW[meta.pillar] / 100) * mults[v.key][reg];
   390→        num += v.dir * w;
   391→        den += w;
   392→      }
   393→      if (den <= 0) continue;
   394→      const net = num / den;
   395→      if (net < 0.15 && net > -0.15) continue;
   396→      const dir: 1 | -1 = net >= 0 ? 1 : -1;
   397→      const agreeing = votes.filter((v) => v.dir === dir);
   398→      if (agreeing.length < 2) continue;
   399→
   400→      const hasAsian = agreeing.some((v) => v.key === "asian_breakout");
   401→      if (hasAsian) {
   402→        const d = dayOf[i];
   403→        if (d === lastAsianDay) continue;
   404→        lastAsianDay = d;
   405→      }
   406→
   407→      const c = candles[i];
   408→      const a = ctx.atrArr[i];
   409→      if (isNaN(a) || a < 0.3) continue;
   410→      const sl = dir === 1 ? c.c - 1.5 * a : c.c + 1.5 * a;
   411→      const sim = simulate(candles, i, dir, sl, hold);
   412→      if (!sim) continue;
   413→      trades.push({
   414→        i,
   415→        t: candles[i].t,
   416→        entryT: candles[i + 1].t,
   417→        dir,
   418→        entry: candles[i + 1].o,
   419→        sl,
   420→        tp1: candles[i + 1].o + dir * Math.abs(candles[i + 1].o - sl),
   421→        result: sim.result,
   422→        r: sim.r,
   423→        bars: sim.bars,
   424→        endIdx: sim.endIdx,
   425→        confidence: clamp(Math.round(Math.abs(net) * 100), 15, 99),
   426→        votes,
   427→        regime: reg,
   428→      });
   429→      busyUntil = sim.endIdx;
   430→    }
   431→    return trades;
   432→  }

   434→  // ---------- 5) الإسناد: من أصاب ومن أخطأ في كل صفقة ----------
   435→  function attribute(trades: TrainerTrade[]): Map<string, { n: number; correct: number }> {
   436→    const map = new Map<string, { n: number; correct: number }>();
   437→    for (const t of trades) {
   438→      for (const v of t.votes) {
   439→        const participated = v.dir === t.dir;
   440→        let correct: boolean | null = null;
   441→        if (t.result === "win") correct = participated;
   442→        else if (t.result === "loss") correct = !participated;
   443→        else if (t.r > 0.05) correct = participated;
   444→        else if (t.r < -0.05) correct = !participated;
   445→        if (correct === null) continue;
   446→        const key = `${v.key}|${t.regime}`;
   447→        const s = map.get(key) ?? { n: 0, correct: 0 };
   448→        s.n++;
   449→        if (correct) s.correct++;
   450→        map.set(key, s);
   451→      }
   452→    }
   453→    return map;
   454→  }

   456→  // ---------- 6) تحديث المضاعفات (التعلم الفعلي) ----------
   457→  function updateMults(
   458→    mults: MultMap,
   459→    attribution: Map<string, { n: number; correct: number }>
   460→  ): { key: string; nameAr: string; regime: Regime; from: number; to: number }[] {
   461→    const changes: { key: string; nameAr: string; regime: Regime; from: number; to: number }[] = [];
   462→    for (const meta of STRATEGY_META) {
   463→      for (const reg of ["trend_up", "trend_down", "range"] as Regime[]) {
   464→        const s = attribution.get(`${meta.key}|${reg}`);
   465→        if (!s || s.n < 6) continue; // عينة صغيرة — لا تعلم
   466→        const target = clamp(wilsonLower(s.correct, s.n) / 0.5, 0.55, 1.45);
   467→        const old = mults[meta.key][reg];
   468→        const next = clamp(old * 0.65 + target * 0.35, 0.5, 1.5);
   469→        if (Math.abs(next - old) >= 0.03) {
   470→          mults[meta.key][reg] = next;
   471→          changes.push({ key: meta.key, nameAr: meta.nameAr, regime: reg, from: old, to: next });
   472→        }
   473→      }
   474→    }
   475→    return changes.sort((a, b) => Math.abs(b.to - b.from) - Math.abs(a.to - a.from));
   476→  }

   478→  // ---------- 7) اشتقاق أوزان الأعمدة من مضاعفات الاستراتيجيات ----------
   479→  function derivePillarWeights(mults: MultMap): Record<string, number> {
   480→    const out: Record<string, number> = {};
   481→    for (const p of PILLAR_KEYS) {
   482→      let num = 0;
   483→      let den = 0;
   484→      for (const meta of STRATEGY_META) {
   485→        if (meta.pillar !== p) continue;
   486→        const m = (mults[meta.key].trend_up + mults[meta.key].trend_down + mults[meta.key].range) / 3;
   487→        num += m;
   488→        den++;
   489→      }
   490→      const pillarMult = den > 0 ? num / den : 1;
   491→      out[p] = BASE_PILLAR_WEIGHTS[p] * pillarMult;
   492→    }
   493→    let sum = PILLAR_KEYS.reduce((a, k) => a + out[k], 0);
   494→    for (const k of PILLAR_KEYS) out[k] = Math.max(6, out[k]);
   495→    sum = PILLAR_KEYS.reduce((a, k) => a + out[k], 0);
   496→    for (const k of PILLAR_KEYS) out[k] = r1((out[k] / sum) * 100);
   497→    const sum2 = PILLAR_KEYS.reduce((a, k) => a + out[k], 0);
   498→    out.trend = r1(out.trend + (100 - sum2)); // تصحيح كسور التقريب
   499→    return out;
   500→  }

   810→      ? highWr - lowWr > 0.08
   811→        ? "معايرة جيدة ✓: توقعات الثقة العالية تتفوق فعلاً على منخفضة الثقة"
   812→        : "الثقة تحتاج معايرة: لا فارق واضح بين مستويات الثقة"
   813→      : "عينات المعايرة غير كافية للحكم";
   814→
   815→  // ---------- 16) ملاحظات الأنظمة السوقية ----------
   816→  const regimeNotes: string[] = [];
   817→  for (const reg of ["trend_up", "trend_down", "range"] as Regime[]) {
   818→    const rt = finalTrades.filter((t) => t.regime === reg);
   819→    if (!rt.length) {
   820→      regimeNotes.push(`${REGIME_AR[reg]}: لم تتولد إشارات كافية`);
   821→      continue;
   822→    }
   823→    const wr = rt.filter((t) => t.result === "win").length / rt.length;
   824→    regimeNotes.push(
   825→      `${REGIME_AR[reg]}: ${rt.length} تنبؤاً — دقة ${r1(wr * 100)}% — ${
   826→        wr >= 0.55 ? "بيئة مواتية لاستراتيجيات البوت" : wr >= 0.45 ? "أداء متوسط" : "بيئة صعبة — البوت يقلص نشاطه فيها"
   827→      }`
   828→    );
   829→  }
   830→
   831→  // ---------- 17) تحليل الشهور (الشهر الحالي + السابق + كل الأشهر) ----------
   832→  const months: BtMonth[] = [];
   833→  {
   834→    interface MonthAgg { ym: string; open: number; close: number; high: number; low: number }
   835→    const byMonth = new Map<string, MonthAgg>();
   836→    const order: string[] = [];
   837→    for (let i = startIdx; i < len; i++) {
   838→      const d = new Date(candles[i].t);
   839→      const ym = `${d.getUTCFullYear()}-${pad2(d.getUTCMonth() + 1)}`;
   840→      let agg = byMonth.get(ym);
   841→      if (!agg) {
   842→        agg = { ym, open: candles[i].o, close: candles[i].c, high: -Infinity, low: Infinity };
   843→        byMonth.set(ym, agg);
   844→        order.push(ym);
   845→      }
   846→      agg.close = candles[i].c;
   847→      agg.high = Math.max(agg.high, candles[i].h);
   848→      agg.low = Math.min(agg.low, candles[i].l);
   849→    }
   850→    for (const ym of order) {
   851→      const agg = byMonth.get(ym)!;
   852→      const mt = finalTrades.filter((t) => {
   853→        const d = new Date(t.t);
   854→        return `${d.getUTCFullYear()}-${pad2(d.getUTCMonth() + 1)}` === ym;
   855→      });
   856→      const mWins = mt.filter((t) => t.result === "win").length;
   857→      const buys = mt.filter((t) => t.dir === 1).length;
   858→      const sells = mt.length - buys;
   859→      const [y, m] = ym.split("-");
   860→      months.push({
   861→        ym,
   862→        label: `${AR_MONTHS[parseInt(m, 10) - 1]} ${y}`,
   863→        open: r1(agg.open),
   864→        high: r1(agg.high),
   865→        low: r1(agg.low),
   866→        close: r1(agg.close),
   867→        changePct: r1(((agg.close - agg.open) / agg.open) * 100),
   868→        atrAvg: 0,
   869→        trades: mt.length,
   870→        wins: mWins,
   871→        wr: mt.length ? r1((mWins / mt.length) * 100) : 0,
   872→        dominant: buys > sells * 1.3 ? "BUY" : sells > buys * 1.3 ? "SELL" : "متوازن",
   873→      });
   874→    }
   875→  }
   876→
   877→  // ---------- 18) المعرفة الذاتية ----------
   878→  const withStats = strategies.filter((s) => s.samples >= 12);
   879→  const bestS = withStats.slice().sort((a, b) => b.winRate - a.winRate)[0] ?? null;
   880→  const worstS = withStats.slice().sort((a, b) => a.winRate - b.winRate)[0] ?? null;
   881→
   882→  if (bestS) {
   883→    lessons.push({
   884→      key: "best",
   885→      text: `أقوى سلاح حالياً: «${bestS.nameAr}» — إصابة ${bestS.winRate}% عبر ${bestS.samples} مشاركة في التوليفة. البوت يرفع وزنها تدريجياً.`,
   886→      count: bestS.samples,
   887→      severity: "good",
   888→    });
   889→  }
   890→  lessons.push({
   891→    key: "calibration",
   892→    text: `${calibVerdict}${
   893→      highN >= 10 && lowN >= 10 ? ` (دقة الثقة العالية ${r1(highWr * 100)}% مقابل ${r1(lowWr * 100)}% للمنخفضة)` : ""
   894→    }.`,
   895→    count: highN,
   896→    severity: highWr > lowWr + 0.08 ? "good" : "info",
   897→  });
   898→  lessons.push({
   899→    key: "holdout",
   900→    text: `الاختبار الأعمى (آخر 25% من الفترة — لم يرها البوت إطلاقاً أثناء التعلم): ${verdict}`,
   901→    count: stAfterH.trades,
   902→    severity: improvement >= 0 ? "good" : "info",
   903→  });
   904→
   905→  const summary = `تدرّب البوت على ${(len - startIdx).toLocaleString("en-US")} شمعة ${
   906→    tf === "1h" ? "ساعة" : "يومية"
   907→  } تغطي نحو 6 أشهر، وأنجز ${finalStats.trades} توقعاً أعمى عبر ${epochsRun} ${
   908→    epochsRun === 1 ? "حلقة تعلم" : "حلقات تعلم"
   909→  }${converged ? " (استقر التعلم مبكراً)" : ""}. دقة التنبؤ على كامل الفترة ${r1(
   910→    finalStats.winRate * 100
   911→  )}%، وفي فترة الاختبار الأعمى ${r1(stAfterH.winRate * 100)}% مقابل ${r1(
   912→    stBeforeH.winRate * 100
   913→  )}% بالأوزان الأساسية قبل التدريب. أقوى استراتيجية: ${
   914→    bestS ? `«${bestS.nameAr}» بإصابة ${bestS.winRate}%` : "غير محددة بعينة كافية"
   915→  }؛ الأضعف: ${worstS ? `«${worstS.nameAr}» بإصابة ${worstS.winRate}%` : "غير محددة"}. استخلص ${
   916→    lessons.length
   917→  } درساً يضبط بها أوزانه المباشرة.`;
   918→
   919→  // ---------- 19) مستوى التدريب ----------
   920→  const epochsPart = Math.min(35, epochsRun * 6);
   921→  const holdoutPart = clamp(improvement * 1.5, -15, 20);
   922→  const sep = highN >= 10 && lowN >= 10 ? (highWr - lowWr) * 100 : 0;
   923→  const calibPart = sep > 5 ? clamp(sep, 5, 15) : 0;
   924→  const coveragePart = finalStats.trades >= 80 ? 10 : Math.round(finalStats.trades / 8);
   925→  const lessonsPart = Math.min(10, lessons.length * 2);
   926→  const score = clamp(Math.round(epochsPart + holdoutPart + calibPart + coveragePart + lessonsPart), 0, 100);
   927→  const label =
   928→    score < 30 ? "مبتدئ" : score < 50 ? "متدرب" : score < 68 ? "متدرب جيداً" : score < 84 ? "متدرب متقدم" : "خبير متدرّب جداً";
   929→
   930→  // ---------- 20) النتيجة النهائية ----------
   931→  return {
   932→    tf,
   933→    epochsRun,
   934→    requestedEpochs: epochsWanted,
   935→    converged,
   936→    generatedAt: new Date().toISOString(),
   937→    dataSource,
   938→    window: {
   939→      from: new Date(candles[startIdx].t).toISOString(),
   940→      to: new Date(candles[len - 1].t).toISOString(),
   941→      candles: len - startIdx,
   942→      months: 6,
   943→    },
   944→    blindStats: finalStats,
   945→    holdout: {
   946→      tradesBefore: stBeforeH.trades,
   947→      winRateBefore: r1(stBeforeH.winRate * 100),
   948→      pfBefore: stBeforeH.profitFactor,
   949→      expBefore: stBeforeH.expectancyR,
   950→      tradesAfter: stAfterH.trades,
   951→      winRateAfter: r1(stAfterH.winRate * 100),
   952→      pfAfter: stAfterH.profitFactor,
   953→      expAfter: stAfterH.expectancyR,
   954→      equityBefore: equityOf(tradesBeforeH).slice(0, 300),
   955→      equityAfter: equityOf(tradesAfterH).slice(0, 300),
   956→      improvementPct: r1(improvement),
   957→      verdict,
   958→    },
   959→    epochLog,
   960→    equity,
   961→    timeline,
   962→    strategies,
   963→    lessons,
   964→    months,
   965→    selfKnowledge: {
   966→      blindAccuracy: r1(finalStats.winRate * 100),
   967→      totalPredictions: finalStats.trades,
   968→      bestStrategy: bestS?.nameAr ?? null,
   969→      bestStrategyWr: bestS?.winRate ?? 0,
   970→      worstStrategy: worstS?.nameAr ?? null,
   971→      worstStrategyWr: worstS?.winRate ?? 0,
   972→      calibration: {
   973→        lowWr: r1(lowWr * 100),
   974→        midWr: r1(midWr * 100),
   975→        highWr: r1(highWr * 100),
   976→        lowN,
   977→        midN,
   978→        highN,
   979→        verdict: calibVerdict,
   980→      },
   981→      regimeNotes,
   982→      summary,
   983→    },
   984→    trainingLevel: { score, label, epochsDone: epochsRun },
   985→    weights: pillarW,
   986→    weightsParam: weightsToParam(pillarW),
   987→  };
   988→}
