     2→// محرك التدريب الذاتي — Walk-Forward Backtest
     3→// البروتوكول (كما طلبه المستخدم):
     4→//   1) البوت "يجرب" كل استراتيجية على شمعة تاريخية C(i)
     5→//      باستخدام بيانات C(0..i) فقط — لا يمكنه رؤية النتيجة
     6→//   2) الدخول على افتتاح الشمعة التالية C(i+1) — بلا نظرة مستقبلية
     7→//   3) بعدها "يتأكد": هل ضرب الهدف أم الوقف؟ (بأولوية الوقف — متحفظ)
     8→//   4) يقوّي استراتيجيته: أوزان الأعمدة تُعدّل وفق Wilson bound
     9→//   5) اختبار إسقاطي صادق: يتعلم على أول 70% ويُختبر على آخر 30%
    10→//      (الفترة الأخيرة لم يرها إطلاقاً أثناء التعلم)
    11→// كل المؤشرات المستخدمة سببية (causal): قيمتها عند i تعتمد فقط
    12→// على الشموع 0..i — إعادة حسابها على الشرائح تعطي نفس النتيجة
    13→// ============================================================

   132→function wilsonLower(wins: number, n: number, z = 1.96): number {
   133→  if (n <= 0) return 0;
   134→  const p = wins / n;
   135→  const denom = 1 + (z * z) / n;
   136→  const center = p + (z * z) / (2 * n);
   137→  const spread = z * Math.sqrt((p * (1 - p)) / n + (z * z) / (4 * n * n));
   138→  return Math.max(0, (center - spread) / denom);
   139→}

   156→function statsOf(trades: RawTrade[]): BtStats {
   157→  const wins = trades.filter((t) => t.result === "win");
   158→  const losses = trades.filter((t) => t.result === "loss");
   159→  const timeouts = trades.filter((t) => t.result === "timeout");
   160→  const grossR = trades.reduce((a, t) => a + t.r, 0);
   161→  const posR = trades.filter((t) => t.r > 0).reduce((a, t) => a + t.r, 0);
   162→  const negR = Math.abs(trades.filter((t) => t.r < 0).reduce((a, t) => a + t.r, 0));
   163→
   164→  // أقصى تراجع تراكمي
   165→  let cum = 0;
   166→  let peak = 0;
   167→  let maxDd = 0;
   168→  for (const t of trades) {
   169→    cum += t.r;
   170→    peak = Math.max(peak, cum);
   171→    maxDd = Math.max(maxDd, peak - cum);
   172→  }
   173→
   174→  // السلاسل
   175→  let best = 0;
   176→  let worst = 0;
   177→  let curW = 0;
   178→  let curL = 0;
   179→  for (const t of trades) {
   180→    if (t.result === "win") {
   181→      curW++;
   182→      curL = 0;
   183→    } else {
   184→      curL++;
   185→      curW = 0;
   186→    }
   187→    best = Math.max(best, curW);
   188→    worst = Math.max(worst, curL);
   189→  }
   190→
   191→  const n = trades.length;
   192→  return {
   193→    trades: n,
   194→    wins: wins.length,
   195→    losses: losses.length,
   196→    timeouts: timeouts.length,
   197→    winRate: n ? wins.length / n : 0,
   198→    avgR: n ? grossR / n : 0,
   199→    grossR: Math.round(grossR * 100) / 100,
   200→    profitFactor: negR > 0 ? Math.round((posR / negR) * 100) / 100 : posR > 0 ? 99 : 0,
   201→    expectancyR: n ? Math.round((grossR / n) * 1000) / 1000 : 0,
   202→    maxDrawdownR: Math.round(maxDd * 100) / 100,
   203→    bestStreak: best,
   204→    worstStreak: worst,
   205→  };
   206→}

   208→/** محاكاة النتيجة — الوقف له الأولوية داخل الشمعة (متحفظ) */
   209→function simulate(
   210→  candles: Candle[],
   211→  i: number,
   212→  dir: 1 | -1,
   213→  sl: number,
   214→  hold: number
   215→): { result: "win" | "loss" | "timeout"; r: number; bars: number; endIdx: number } | null {
   216→  const next = candles[i + 1];
   217→  if (!next) return null;
   218→  const entry = next.o;
   219→  const risk = Math.abs(entry - sl);
   220→  if (risk < 0.3) return null; // وقف غير منطقي — تجاهل
   221→  const tp1 = entry + dir * risk;
   222→
   223→  const last = Math.min(i + hold, candles.length - 1);
   224→  for (let j = i + 1; j <= last; j++) {
   225→    const c = candles[j];
   226→    if (dir === 1) {
   227→      if (c.l <= sl) return { result: "loss", r: -1, bars: j - i, endIdx: j };
   228→      if (c.h >= tp1) return { result: "win", r: 1, bars: j - i, endIdx: j };
   229→    } else {
   230→      if (c.h >= sl) return { result: "loss", r: -1, bars: j - i, endIdx: j };
   231→      if (c.l <= tp1) return { result: "win", r: 1, bars: j - i, endIdx: j };
   232→    }
   233→  }
   234→  const exit = candles[last].c;
   235→  const r = ((exit - entry) * dir) / risk;
   236→  return { result: "timeout", r: Math.round(r * 100) / 100, bars: last - i, endIdx: last };
   237→}

   252→function buildDays(candles: Candle[]): { days: DayInfo[]; dayOf: number[] } {
   253→  const days: DayInfo[] = [];
   254→  const dayOf: number[] = new Array(candles.length).fill(-1);
   255→  const map = new Map<string, DayInfo>();
   256→
   257→  for (let i = 0; i < candles.length; i++) {
   258→    const d = new Date(candles[i].t);
   259→    const key = `${d.getUTCFullYear()}-${String(d.getUTCMonth() + 1).padStart(2, "0")}-${String(d.getUTCDate()).padStart(2, "0")}`;
   260→    let info = map.get(key);
   261→    if (!info) {
   262→      info = {
   263→        idx: days.length,
   264→        key,
   265→        high: -Infinity,
   266→        low: Infinity,
   267→        asianHigh: -Infinity,
   268→        asianLow: Infinity,
   269→        asianCount: 0,
   270→        firstIdx: i,
   271→        lastIdx: i,
   272→      };
   273→      map.set(key, info);
   274→      days.push(info);
   275→    }
   276→    info.high = Math.max(info.high, candles[i].h);
   277→    info.low = Math.min(info.low, candles[i].l);
   278→    info.lastIdx = i;
   279→    const h = d.getUTCHours();
   280→    if (h < 8) {
   281→      info.asianHigh = Math.max(info.asianHigh, candles[i].h);
   282→      info.asianLow = Math.min(info.asianLow, candles[i].l);
   283→      info.asianCount++;
   284→    }
   285→    dayOf[i] = info.idx;
   286→  }
   287→  return { days, dayOf };
   288→}

   318→function evaluateAt(ctx: Ctx, i: number): Vote[] {
   319→  const votes: Vote[] = [];
   320→  const { candles, closes } = ctx;
   321→  const c = candles[i];
   322→  const p = candles[i - 1];
   323→  const a = ctx.atrArr[i];
   324→  if (!a || a < 0.3 || isNaN(a)) return votes;
   325→
   326→  const body = Math.abs(c.c - c.o);
   327→  const e9 = ctx.ema9[i];
   328→  const e21 = ctx.ema21[i];
   329→  const e50 = ctx.ema50[i];
   330→  const e200 = ctx.ema200[i];
   331→  const rsiNow = ctx.rsiArr[i];
   332→  const rsiPrev = ctx.rsiArr[i - 1];
   333→
   334→  const ok = (...vals: number[]) => vals.every((v) => v != null && !isNaN(v));
   335→
   336→  // 1) ema_trend: ترتيب كامل + ADX ≥ 20
   337→  if (ok(e9, e21, e50, ctx.adxArr[i]) && ctx.adxArr[i] >= 20) {
   338→    if (e9 > e21 && e21 > e50 && c.c > e9) votes.push({ key: "ema_trend", dir: 1, sl: c.c - 1.5 * a });
   339→    if (e9 < e21 && e21 < e50 && c.c < e9) votes.push({ key: "ema_trend", dir: -1, sl: c.c + 1.5 * a });
   340→  }
   341→
   342→  // 2) macd_cross: تقاطع خلال آخر شمعتين + اتجاه EMA50
   343→  const ml = ctx.macdLine[i];
   344→  const ms = ctx.macdSignal[i];
   345→  const mlP = ctx.macdLine[i - 1];
   346→  const msP = ctx.macdSignal[i - 1];
   347→  if (ok(ml, ms, mlP, msP, e50)) {
   348→    if (ml > ms && mlP <= msP && c.c > e50) votes.push({ key: "macd_cross", dir: 1, sl: c.c - 1.5 * a });
   349→    if (ml < ms && mlP >= msP && c.c < e50) votes.push({ key: "macd_cross", dir: -1, sl: c.c + 1.5 * a });
   350→  }
   351→
   352→  // 3) bb_break: انضغاط سابق + إغلاق خارج الشريط بجسم قوي
   353→  const bwNow = ctx.bw[i];
   354→  const bwAvgNow = ctx.bwAvg[i];
   355→  const squeezePrev = ok(ctx.bw[i - 1], ctx.bwAvg[i - 1]) && ctx.bw[i - 1] < 0.8 * ctx.bwAvg[i - 1];
   356→  if (squeezePrev && ok(ctx.bbUpper[i], ctx.bbLower[i]) && body > 0.5 * a) {
   357→    if (c.c > ctx.bbUpper[i]) votes.push({ key: "bb_break", dir: 1, sl: c.c - 1.5 * a });
   358→    if (c.c < ctx.bbLower[i]) votes.push({ key: "bb_break", dir: -1, sl: c.c + 1.5 * a });
   359→  }
   360→  void bwNow;
   361→  void bwAvgNow;
   362→
   363→  // 4) rsi_reversal: خروج من التطرف + مع EMA200
   364→  if (ok(rsiNow, rsiPrev, e200)) {
   365→    if (rsiPrev < 32 && rsiNow > rsiPrev && c.c > e200)
   366→      votes.push({ key: "rsi_reversal", dir: 1, sl: c.c - 1.5 * a });
   367→    if (rsiPrev > 68 && rsiNow < rsiPrev && c.c < e200)
   368→      votes.push({ key: "rsi_reversal", dir: -1, sl: c.c + 1.5 * a });
   369→  }
   370→
   371→  // 5) bb_reversal: اختراق الشريط ثم عودة داخلية + RSI
   372→  const up = ctx.bbUpper[i];
   373→  const lo = ctx.bbLower[i];
   374→  const upP = ctx.bbUpper[i - 1];
   375→  const loP = ctx.bbLower[i - 1];
   376→  if (ok(up, lo, upP, loP, rsiNow)) {
   377→    if (p.l < loP && c.c > lo && rsiNow < 42)
   378→      votes.push({ key: "bb_reversal", dir: 1, sl: c.c - 1.5 * a });
   379→    if (p.h > upP && c.c < up && rsiNow > 58)
   380→      votes.push({ key: "bb_reversal", dir: -1, sl: c.c + 1.5 * a });
   381→  }
   382→
   383→  // 6) sr_bounce: دعم/مقاومة ديناميكي (آخر 12 شمعة سابقة)
   384→  let support = Infinity;
   385→  let resistance = -Infinity;
   386→  for (let j = i - 12; j < i; j++) {
   387→    if (j < 0) continue;
   388→    support = Math.min(support, candles[j].l);
   389→    resistance = Math.max(resistance, candles[j].h);
   390→  }
   391→  if (ok(support, resistance) && isFinite(support) && isFinite(resistance)) {
   392→    if (c.l <= support + 0.25 * a && c.c > c.o && c.c > support)
   393→      votes.push({ key: "sr_bounce", dir: 1, sl: c.c - 1.2 * a });
   394→    if (c.h >= resistance - 0.25 * a && c.c < c.o && c.c < resistance)
   395→      votes.push({ key: "sr_bounce", dir: -1, sl: c.c + 1.2 * a });
   396→  }
   397→
   398→  // 7) smc_sweep: كنس قاع/قمة اليوم السابق ثم استعادة
   399→  const dayIdx = ctx.dayOf[i];
   400→  if (dayIdx >= 1) {
   401→    const prevDay = ctx.days[dayIdx - 1];
   402→    if (prevDay && prevDay.lastIdx < i) {
   403→      if (c.l < prevDay.low && c.c > prevDay.low && c.c > c.o) {
   404→        const slDist = clamp(c.c - c.l + 0.25 * a, 0.8 * a, 2.5 * a);
   405→        votes.push({ key: "smc_sweep", dir: 1, sl: c.c - slDist });
   406→      }
   407→      if (c.h > prevDay.high && c.c < prevDay.high && c.c < c.o) {
   408→        const slDist = clamp(c.h - c.c + 0.25 * a, 0.8 * a, 2.5 * a);
   409→        votes.push({ key: "smc_sweep", dir: -1, sl: c.c + slDist });
   410→      }
   411→    }
   412→  }
   413→
   414→  // 8) pattern_play: ابتلاع/مطرقة/شهاب مع فلتر EMA200
   415→  const bodyP = c.h - c.l > 0 ? body / (c.h - c.l) : 0;
   416→  const upperWick = c.h - Math.max(c.c, c.o);
   417→  const lowerWick = Math.min(c.c, c.o) - c.l;
   418→  const prevBody = Math.abs(p.c - p.o);
   419→  if (ok(e200) && c.h - c.l > 0) {
   420→    const bullEngulf = p.c < p.o && c.c > c.o && c.c >= p.o && c.o <= p.c && body > prevBody;
   421→    const bearEngulf = p.c > p.o && c.c < c.o && c.o >= p.c && c.c <= p.o && body > prevBody;
   422→    const hammer = lowerWick > body * 2 && upperWick < body * 0.8 && bodyP < 0.45;
   423→    const star = upperWick > body * 2 && lowerWick < body * 0.8 && bodyP < 0.45;
   424→    if ((bullEngulf || (hammer && c.c > c.o)) && c.c > e200)
   425→      votes.push({ key: "pattern_play", dir: 1, sl: c.c - 1.2 * a });
   426→    if ((bearEngulf || (star && c.c < c.o)) && c.c < e200)
   427→      votes.push({ key: "pattern_play", dir: -1, sl: c.c + 1.2 * a });
   428→  }
   429→
   430→  // 9) asian_breakout: كسر نطاق آسيوي بعد 07:00 UTC
   431→  const dayInfo = ctx.days[dayIdx];
   432→  const hourUtc = new Date(c.t).getUTCHours();
   433→  if (dayInfo && dayInfo.asianCount >= 4 && hourUtc >= 7 && isFinite(dayInfo.asianHigh) && isFinite(dayInfo.asianLow)) {
   434→    const rangeWidth = dayInfo.asianHigh - dayInfo.asianLow;
   435→    const onlyAfterLondon = hourUtc >= 7 && hourUtc <= 16; // نافذة لندن فقط
   436→    if (onlyAfterLondon && body > 0.6 * a) {
   437→      if (c.c > dayInfo.asianHigh) {
   438→        const slDist = Math.max(1.0 * a, rangeWidth * 0.5);
   439→        votes.push({ key: "asian_breakout", dir: 1, sl: c.c - slDist });
   440→      }
   441→      if (c.c < dayInfo.asianLow) {
   442→        const slDist = Math.max(1.0 * a, rangeWidth * 0.5);
   443→        votes.push({ key: "asian_breakout", dir: -1, sl: c.c + slDist });
   444→      }
   445→    }
   446→  }
   447→
   448→  return votes;
   449→}

   451→// ---------- التعلّم ----------
   452→function learnWeights(
   453→  stratTrades: Map<string, RawTrade[]>,
   454→  base: Record<string, number>
   455→): Record<string, number> {
   456→  const mult: Record<string, number> = {};
   457→  for (const meta of STRATEGY_META) {
   458→    const trades = stratTrades.get(meta.key) ?? [];
   459→    const n = trades.length;
   460→    if (n >= 12) {
   461→      const wins = trades.filter((t) => t.result === "win").length;
   462→      const wl = wilsonLower(wins, n);
   463→      mult[meta.key] = clamp(wl / 0.5, 0.55, 1.45);
   464→    } else {
   465→      mult[meta.key] = 1; // عينة صغيرة — لا تعلّم
   466→    }
   467→  }
   468→
   469→  // مضاعف العمود = متوسط مرجّح بعدد صفقات استراتيجياته
   470→  const pillarMult: Record<string, number> = {};
   471→  for (const pKey of PILLAR_KEYS) {
   472→    let num = 0;
   473→    let den = 0;
   474→    for (const meta of STRATEGY_META) {
   475→      if (meta.pillar !== pKey) continue;
   476→      const n = (stratTrades.get(meta.key) ?? []).length;
   477→      num += mult[meta.key] * n;
   478→      den += n;
   479→    }
   480→    pillarMult[pKey] = den > 0 ? num / den : 1;
   481→  }
   482→
   483→  const learned: Record<string, number> = {};
   484→  for (const pKey of PILLAR_KEYS) {
   485→    learned[pKey] = base[pKey] * pillarMult[pKey];
   486→  }
   487→  // إعادة التطبيع إلى 100 مع حد أدنى 6
   488→  let sum = PILLAR_KEYS.reduce((a, k) => a + learned[k], 0);
   489→  for (const k of PILLAR_KEYS) learned[k] = Math.max(6, learned[k]);
   490→  sum = PILLAR_KEYS.reduce((a, k) => a + learned[k], 0);
   491→  for (const k of PILLAR_KEYS) learned[k] = Math.round((learned[k] / sum) * 1000) / 10;
   492→  return learned;
   493→}

   504→  }

   506→  const closes = candles.map((c) => c.c);

   519→    const u = bbData.upper[i];
   520→    const m = bbData.mid[i];
   521→    if (!isNaN(u) && !isNaN(m) && m > 0) bw[i] = ((bbData.upper[i] - bbData.lower[i]) / m) * 100;
   522→  }
   523→  const bwAvg = rollingAvg(bw, 50);
   524→

   526→
   527→  const ctx: Ctx = {
   528→    candles,
   529→    closes,
   530→    ema9: emaSet.ema9,
   531→    ema21: emaSet.ema21,
   532→    ema50: emaSet.ema50,
   533→    ema200: emaSet.ema200,
   534→    rsiArr,
   535→    macdLine: macdData.line,
   536→    macdSignal: macdData.signal,
   537→    macdHist: macdData.histogram,
   538→    bbUpper: bbData.upper,
   539→    bbLower: bbData.lower,
   540→    bw,
   541→    bwAvg,
   542→    atrArr: atrData.values,
   543→    adxArr,
   544→    days,
   545→    dayOf,
   546→  };
   547→
   548→  const len = candles.length;
   549→  const split = WARMUP + Math.floor((len - WARMUP - hold) * 0.7); // 70% تدريب / 30% اختبار
   550→
   551→  // ===== 1) تقييم كل الشموع (التصويت) — مرة واحدة =====
   552→  const votesAll: Vote[][] = new Array(len);
   553→  for (let i = WARMUP; i < len - 1; i++) {
   554→    votesAll[i] = evaluateAt(ctx, i);
   555→  }
   556→
   557→  // ===== 2) صفقات كل استراتيجية (مع تهدئة — صفقة واحدة نشطة) =====
   558→  const stratTrades = new Map<string, RawTrade[]>();
   559→  const takenPerStrategy = new Map<string, number[]>(); // فهرس الشموع المأخوذة
   560→  for (const meta of STRATEGY_META) {
   561→    const trades: RawTrade[] = [];
   562→    const taken: number[] = [];
   563→    let busyUntil = -1;
   564→    let lastAsianDay = -1; // كسر آسيوي واحد لكل يوم
   565→    for (let i = WARMUP; i < len - 1; i++) {
   566→      const votes = votesAll[i] ?? [];
   567→      const vote = votes.find((v) => v.key === meta.key);

   573→        lastAsianDay = d;
   574→      }

   579→        i,
   580→        t: candles[i].t,

   586→        result: sim.result,

   598→  // ===== 3) التوليفة المركّبة (إشارة البوت الكلية) =====
   599→  function runComposite(
   600→    from: number,
   601→    to: number,
   602→    weights: Record<string, number> | null // null = أوزان متساوية
   603→  ): RawTrade[] {
   604→    const trades: RawTrade[] = [];
   605→    let busyUntil = -1;
   606→    let lastAsianDay = -1;
   607→    for (let i = from; i < to; i++) {
   608→      const votes = votesAll[i] ?? [];
   609→      if (!votes.length) continue;
   610→      if (i <= busyUntil) continue;
   611→      let num = 0;
   612→      let den = 0;
   613→      for (const v of votes) {
   614→        const meta = STRATEGY_META.find((m) => m.key === v.key)!;
   615→        const w = weights ? weights[meta.pillar] : 1;
   616→        num += v.dir * w;
   617→        den += w;
   618→      }
   619→      if (den === 0) continue;
   620→      const net = num / den;
   621→      if (net < 0.15 && net > -0.15) continue;
   622→      const dir: 1 | -1 = net >= 0.15 ? 1 : -1;
   623→
   624→      const agreeing = votes.filter((v) => v.dir === dir);
   625→      if (agreeing.length < 2) continue; // توافق صورتين على الأقل
   626→

   629→      if (hasAsian) {
   630→        const d = dayOf[i];
   631→        if (d === lastAsianDay) continue;
   632→        lastAsianDay = d;
   633→      }
   634→
   635→      const c = candles[i];
   636→      const a = ctx.atrArr[i];
   637→      const sl = dir === 1 ? c.c - 1.5 * a : c.c + 1.5 * a;
   638→      const sim = simulate(candles, i, dir, sl, hold);
   639→      if (!sim) continue;
   640→      trades.push({
   641→        key: "composite",
   642→        i,
   643→        t: candles[i].t,
   644→        entryT: candles[i + 1].t,
   645→        dir,
   646→        entry: candles[i + 1].o,
   647→        sl,
   648→        tp1: candles[i + 1].o + dir * Math.abs(candles[i + 1].o - sl),

   650→        r: sim.r,
   651→        bars: sim.bars,
   652→        endIdx: sim.endIdx,

   662→
   663→  const base = { ...BASE_PILLAR_WEIGHTS } as Record<string, number>;

   665→
   666→  const tradesBeforeTest = runComposite(split, len - 1, null); // أوزان متساوية

   670→  // أوزان التطبيق المباشر — تعلم من كامل التاريخ

   672→  const learnedLiveW = learnWeights(fullStratTrades, base);

   674→
   675→  // حفظ حالة التعلّم ليمتد إليها محرك الإشارة المباشر
   676→  setLearnedState({
   677→    weights: learnedLiveW,
   678→    stats: {

   713→      multiplier: n >= 12 ? Math.round(clamp(wl / 0.5, 0.55, 1.45) * 100) / 100 : 1,

   846→// ---------- متوسط متحرك سببي ----------
   847→function rollingAvg(values: number[], win: number): number[] {
   848→  const out: number[] = new Array(values.length).fill(NaN);
   849→  let sum = 0;
   850→  let count = 0;
   851→  const queue: number[] = [];
   852→  for (let i = 0; i < values.length; i++) {
   853→    const v = values[i];
   854→    if (!isNaN(v)) {
   855→      queue.push(v);
   856→      sum += v;
   857→      count++;
   858→      if (queue.length > win) {
   859→        sum -= queue.shift()!;
   860→        count--;
   861→      }
   862→    }
   863→    out[i] = count >= Math.min(win, 20) ? sum / count : NaN;
   864→  }
   865→  return out;
   866→}

   868→// ---------- سلسلة ADX سببية (Wilder) ----------
   869→function adxSeries(candles: Candle[], period: number): number[] {
   870→  const n = candles.length;
   871→  const out: number[] = new Array(n).fill(NaN);
   872→  if (n < period * 2 + 1) return out;
   873→
   874→  const plusDM: number[] = [0];
   875→  const minusDM: number[] = [0];
   876→  const trs: number[] = [candles[0].h - candles[0].l];
   877→  for (let i = 1; i < n; i++) {
   878→    const upMove = candles[i].h - candles[i - 1].h;
   879→    const downMove = candles[i - 1].l - candles[i].l;
   880→    plusDM.push(upMove > downMove && upMove > 0 ? upMove : 0);
   881→    minusDM.push(downMove > upMove && downMove > 0 ? downMove : 0);
   882→    const prevClose = candles[i - 1].c;
   883→    trs.push(Math.max(candles[i].h - candles[i].l, Math.abs(candles[i].h - prevClose), Math.abs(candles[i].l - prevClose)));
   884→  }
   885→
   886→  const smooth = (arr: number[]): number[] => {
   887→    const s: number[] = new Array(arr.length).fill(NaN);
   888→    let sum = 0;
   889→    for (let i = 0; i < period; i++) sum += arr[i];
   890→    s[period - 1] = sum;
   891→    for (let i = period; i < arr.length; i++) s[i] = s[i - 1] - s[i - 1] / period + arr[i];
   892→    return s;
   893→  };

   897→  const smMDM = smooth(minusDM);

   910→  }
